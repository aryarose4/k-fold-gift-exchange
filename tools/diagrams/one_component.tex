% Source for the "one connected component" tikz-cd diagram.
% Six names arranged as a hexagon on a 4-column, 3-row grid:
%   row1: cols 2-3, row2: cols 1 & 4, row3: cols 2-3.
% Each person has exactly ONE outgoing arrow; the arrows form a single
% 6-cycle around the perimeter (one connected component).
% CDCOLOR below is the RGB triplet of the stroke/label color, substituted
% at build time (light vs dark theme). See build.sh.
% column sep / row sep are LARGE to match two_components.tex.
\documentclass{article}
\usepackage[margin=0pt]{geometry}
\usepackage{amsmath}
\usepackage{xcolor}
\usepackage{tikz-cd}
\definecolor{cdcolor}{RGB}{CDCOLOR}
\begin{document}
\thispagestyle{empty}
\begin{tikzcd}[column sep = 3em, row sep = 10 ex, arrows = {-stealth, color = cdcolor}]
  & \textcolor{cdcolor}{\text{Alice}} \arrow[from=1-2, to=3-3]
  & \textcolor{cdcolor}{\text{Bob}} \arrow[from=1-3, to=1-2] & \\
  \textcolor{cdcolor}{\text{Frank}} \arrow[from=2-1, to=1-3]
  & & & \textcolor{cdcolor}{\text{Carlos}} \arrow[from=2-4, to=2-1] \\
  & \textcolor{cdcolor}{\text{Eve}} \arrow[from=3-2, to=2-4]
  & \textcolor{cdcolor}{\text{David}} \arrow[from=3-3, to=3-2] &
\end{tikzcd}
\end{document}
